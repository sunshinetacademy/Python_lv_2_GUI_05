بدل ما نطبع رسالة في الكونسول والمستخدم ما يشوفهاش، بنطلّع نافذة صغيرة فوق البرنامج فيها رسالة وأزرار.

messagebox_demo.py
import tkinter as tk
from tkinter import messagebox

root = tk.Tk()
root.title("الرسائل 💬")
root.geometry("420x300")


def say_info():
    # ℹ️ رسالة معلومات — زر OK واحد
    messagebox.showinfo("أهلاً", "طلبك وصل للمطبخ! 🍕")


def say_warning():
    # ⚠️ تحذير — مثلث أصفر
    messagebox.showwarning("تنبيه", "الجبنة خلصت!")


def say_error():
    # ❌ خطأ — دائرة حمراء
    messagebox.showerror("خطأ", "الكمية لازم رقم!")


def confirm_delete():
    # ❓ سؤال — بيرجع True أو False
    answer = messagebox.askyesno("تأكيد", "تلغي الطلب؟")
    if answer:
        status.config(text="🗑️ اتلغى الطلب", fg="red")
    else:
        status.config(text="✅ الطلب لسه شغال", fg="green")


tk.Button(root, text="ℹ️ معلومات", command=say_info).pack(pady=6, fill="x", padx=30)
tk.Button(root, text="⚠️ تحذير", command=say_warning).pack(pady=6, fill="x", padx=30)
tk.Button(root, text="❌ خطأ", command=say_error).pack(pady=6, fill="x", padx=30)
tk.Button(root, text="❓ تأكيد الإلغاء", command=confirm_delete).pack(pady=6, fill="x", padx=30)

status = tk.Label(root, text="جاهز", font=("Arial", 12, "bold"))
status.pack(pady=14)

root.mainloop()

radiobutton_demo.py
import tkinter as tk

root = tk.Tk()
root.title("اختيار الحجم 🔘")
root.geometry("400x320")

# 🔑 متغير واحد مشترك بين كل الأزرار
size_var = tk.StringVar(value="medium")

# قاموس الأسعار
PRICES = {"small": 60, "medium": 90, "large": 130}


def on_size_change():
    choice = size_var.get()
    result.config(text=f"الحجم: {choice} — {PRICES[choice]} جنيه")


tk.Label(root, text="🍕 اختار حجم البيتزا:",
         font=("Arial", 14, "bold")).pack(pady=14)

# كل زر: نفس variable + value مختلف
for text, val in [("🍕 صغير", "small"),
                  ("🍕 وسط", "medium"),
                  ("🍕 كبير", "large")]:
    tk.Radiobutton(
        root,
        text=text,
        variable=size_var,     # ← نفس المتغير!
        value=val,             # ← قيمة مختلفة
        command=on_size_change,
        font=("Arial", 12),
        indicatoron=True,
    ).pack(anchor="w", padx=60, pady=4)

result = tk.Label(root, text="الحجم: medium — 90 جنيه",
                 font=("Arial", 12, "bold"), fg="#dc2626")
result.pack(pady=18)

root.mainloop()

الإضافات! جبنة زيادة + زيتون + فلفل — كله ممكن مع بعض. هنا كل مربع ليه متغير مستقل من نوع BooleanVar.

checkbutton_demo.py
import tkinter as tk

root = tk.Tk()
root.title("الإضافات ☑️")
root.geometry("420x380")

# 🔑 متغير مستقل لكل مربع (عكس Radiobutton!)
extras = {
    "جبنة زيادة 🧀": (tk.BooleanVar(value=False), 15),
    "زيتون 🫒":      (tk.BooleanVar(value=False), 10),
    "فلفل 🌶️":       (tk.BooleanVar(value=False), 8),
    "مشروم 🍄":      (tk.BooleanVar(value=False), 12),
}


def calculate():
    total = 0
    chosen = []
    for name, (var, price) in extras.items():
        if var.get():              # True يعني متعلّم عليه
            chosen.append(name)
            total += price

    if chosen:
        result.config(text=" + ".join(chosen) + f"\n💰 {total} جنيه")
    else:
        result.config(text="مفيش إضافات — 0 جنيه")


tk.Label(root, text="🍕 اختار إضافاتك:",
         font=("Arial", 14, "bold")).pack(pady=14)

for name, (var, price) in extras.items():
    tk.Checkbutton(
        root,
        text=f"{name}  (+{price} جنيه)",
        variable=var,          # ← متغير خاص بيه
        onvalue=True,
        offvalue=False,
        command=calculate,
        font=("Arial", 12),
    ).pack(anchor="w", padx=50, pady=3)

result = tk.Label(root, text="مفيش إضافات — 0 جنيه",
                 font=("Arial", 12, "bold"), fg="#15803d", justify="right")
result.pack(pady=18)

root.mainloop()

import tkinter as tk
from tkinter import ttk, messagebox

BG    = "#fff7ed"
CARD  = "#ffffff"
INK   = "#2b1d14"
CRUST = "#f59e0b"
TOM   = "#dc2626"
BASIL = "#15803d"

SIZE_PRICES  = {"small": 60, "medium": 90, "large": 130}
SIZE_NAMES   = {"small": "صغير", "medium": "وسط", "large": "كبير"}
CRUST_EXTRA  = {"عادي": 0, "رقيقة": 0, "سميكة": 0, "محشية أطراف": 20}
DRINK_PRICES = {"بدون": 0, "كولا": 20, "عصير": 25, "مياه": 10}
EXTRA_PRICES = {"جبنة زيادة": 15, "زيتون": 10, "فلفل": 8,
                "مشروم": 12, "سلامي": 20}

root = tk.Tk()
root.title("🍕 Sunshine Pizza — Sunshine T Academy")
root.geometry("520x720")
root.minsize(500, 660)
root.configure(bg=BG)

# ═══ المتغيرات ═══
name_var   = tk.StringVar(value="")
size_var   = tk.StringVar(value="medium")   # Radio: واحد مشترك
crust_var  = tk.StringVar(value="عادي")
drink_var  = tk.StringVar(value="بدون")
count_var  = tk.IntVar(value=1)
coupon_var = tk.StringVar(value="")
sauce_var  = tk.BooleanVar(value=False)
deliv_var  = tk.BooleanVar(value=False)

# Check: متغير مستقل لكل إضافة
extra_vars = {name: tk.BooleanVar(value=False) for name in EXTRA_PRICES}


def calculate():
    """حساب الفاتورة كاملة"""
    unit = SIZE_PRICES[size_var.get()]
    unit += CRUST_EXTRA[crust_var.get()]
    if sauce_var.get():
        unit += 5
    for name, var in extra_vars.items():
        if var.get():
            unit += EXTRA_PRICES[name]

    pizzas = unit * count_var.get()
    drink  = DRINK_PRICES[drink_var.get()]
    deliv  = 25 if deliv_var.get() else 0
    sub    = pizzas + drink + deliv

    discount = 0
    if coupon_var.get().strip().upper() == "PIZZA10":
        discount = sub * 0.10

    return unit, pizzas, drink, deliv, sub, discount, sub - discount


def refresh(*args):
    unit, pizzas, drink, deliv, sub, disc, total = calculate()
    lines = [f"🍕 سعر البيتزا: {unit} جنيه",
             f"🔢 العدد: {count_var.get()}",
             f"🍕 إجمالي البيتزا: {pizzas} جنيه"]
    if drink:
        lines.append(f"🥤 المشروب: {drink} جنيه")
    if deliv:
        lines.append(f"🛵 التوصيل: {deliv} جنيه")
    lines.append(f"المجموع: {sub:.0f} جنيه")
    if disc:
        lines.append(f"🎉 خصم 10%: -{disc:.0f} جنيه")
    lines.append(f"💰 الإجمالي: {total:.0f} جنيه")
    receipt.config(text="\n".join(lines))


def place_order():
    name = name_var.get().strip()
    if not name:
        messagebox.showwarning("ناقص بيانات", "⚠️ اكتب اسمك الأول!")
        name_entry.focus_set()
        return

    unit, pizzas, drink, deliv, sub, disc, total = calculate()
    chosen = [n for n, v in extra_vars.items() if v.get()]
    extras_txt = "، ".join(chosen) if chosen else "بدون إضافات"

    summary = (
        f"👤 الاسم: {name}\n"
        f"📏 الحجم: {SIZE_NAMES[size_var.get()]}\n"
        f"🥖 العجينة: {crust_var.get()}\n"
        f"🧀 الإضافات: {extras_txt}\n"
        f"🥤 المشروب: {drink_var.get()}\n"
        f"🔢 العدد: {count_var.get()}\n"
        f"💰 الإجمالي: {total:.0f} جنيه\n\n"
        "تأكيد الطلب؟"
    )

    # ❓ رسالة التأكيد — القرار بيد المستخدم
    if messagebox.askyesno("تأكيد الطلب 🍕", summary):
        messagebox.showinfo("تم الطلب ✅",
                              f"طلبك في الفرن يا {name}!\n"
                              f"المبلغ: {total:.0f} جنيه\n"
                              "⏱️ التوصيل خلال 30 دقيقة")
        status.config(text="✅ تم إرسال الطلب للمطبخ", fg=BASIL)
    else:
        messagebox.showwarning("إلغاء", "تم إلغاء الطلب — عدّل واختار تاني.")
        status.config(text="⚠️ الطلب اتلغى", fg=TOM)


def clear_all():
    if messagebox.askokcancel("تفريغ", "تمسح كل الاختيارات؟"):
        name_var.set("")
        size_var.set("medium")
        crust_var.set("عادي")
        drink_var.set("بدون")
        count_var.set(1)
        coupon_var.set("")
        sauce_var.set(False)
        deliv_var.set(False)
        for v in extra_vars.values():
            v.set(False)
        status.config(text="🧹 اتفضّى الطلب", fg=INK)
